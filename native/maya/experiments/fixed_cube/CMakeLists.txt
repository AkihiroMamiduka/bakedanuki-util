cmake_minimum_required(VERSION 3.27.3)
project(bd_fixed_cube_research LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR MSVC_VERSION LESS 1938)
    message(FATAL_ERROR "Windows x64 / Visual Studio 2022 17.8.3 or later is required.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Maya plug-ins must be built for a 64-bit target.")
endif()
set(MAYA_VERSION "2025" CACHE STRING "Target Maya version")
set(MAYA_LOCATION "C:/Program Files/Autodesk/Maya${MAYA_VERSION}" CACHE PATH "Maya installation")
option(BD_CUBE_PROFILE "Build diagnostic callback counters and timers" OFF)
option(BD_CUBE_DRAW_PROFILE "Observe stock shader draw batches (changes measurement overhead)" OFF)
option(BD_CUBE_RESEARCH_FIN_STUDY "Build the two dedicated Fin Cube variants" OFF)
option(BD_CUBE_RESEARCH_DUMMY_BOOL_STUDY "Build the fixed Fin Cube with zero and 100 dummy bools" OFF)
option(BD_CUBE_RESEARCH_STATIC_ATTRIBUTES_STUDY "Build the fixed-value and editable Cube and Fin comparators" OFF)
option(BD_CUBE_RESEARCH_DISPLAY_STATE_STUDY "Build the deformable Cube display-state comparators" OFF)
if(BD_CUBE_DRAW_PROFILE)
    set(BD_CUBE_PROFILE ON)
endif()
set(BD_CUBE_OUTPUT_SUFFIX "" CACHE STRING "Output suffix for an isolated comparison build")
set(BD_CUBE_VARIANT "Current" CACHE STRING "Fixed Cube research implementation")
set_property(CACHE BD_CUBE_VARIANT PROPERTY STRINGS Current NodeParallel SubSceneMatrix Draw)
option(BD_CUBE_RESEARCH_SHAPE_MATRIX "Enable the shape matrix input in a research DLL" OFF)
set(BD_CUBE_ARTIFACT_DIRECTORY "" CACHE PATH "Optional directory for Release research DLLs")
if(NOT BD_CUBE_VARIANT MATCHES "^(Current|NodeParallel|SubSceneMatrix|Draw)$")
    message(FATAL_ERROR "Unknown fixed Cube research variant: ${BD_CUBE_VARIANT}")
endif()
if(BD_CUBE_VARIANT MATCHES "^(SubSceneMatrix|Draw)$" AND BD_CUBE_DRAW_PROFILE)
    message(FATAL_ERROR "Draw profiling is supported by the Geometry research target.")
endif()
set(CUBE_OUTPUT_SUFFIX "${BD_CUBE_OUTPUT_SUFFIX}")
if(CUBE_OUTPUT_SUFFIX STREQUAL "")
    if(BD_CUBE_VARIANT STREQUAL "NodeParallel")
        set(CUBE_OUTPUT_SUFFIX "ResearchParallel")
    elseif(BD_CUBE_VARIANT STREQUAL "SubSceneMatrix")
        set(CUBE_OUTPUT_SUFFIX "ResearchSubScene")
    elseif(BD_CUBE_VARIANT STREQUAL "Draw")
        set(CUBE_OUTPUT_SUFFIX "ResearchDraw")
    endif()
    if(BD_CUBE_RESEARCH_SHAPE_MATRIX)
        if(CUBE_OUTPUT_SUFFIX STREQUAL "")
            set(CUBE_OUTPUT_SUFFIX "ResearchShapeMatrix")
        else()
            string(APPEND CUBE_OUTPUT_SUFFIX "ShapeMatrix")
        endif()
    endif()
    if(BD_CUBE_DRAW_PROFILE)
        if(CUBE_OUTPUT_SUFFIX STREQUAL "")
            set(CUBE_OUTPUT_SUFFIX "ResearchDrawProfile")
        else()
            string(APPEND CUBE_OUTPUT_SUFFIX "DrawProfile")
        endif()
    elseif(BD_CUBE_PROFILE)
        if(CUBE_OUTPUT_SUFFIX STREQUAL "")
            set(CUBE_OUTPUT_SUFFIX "ResearchProfile")
        else()
            string(APPEND CUBE_OUTPUT_SUFFIX "Profile")
        endif()
    endif()
endif()
if(BD_CUBE_DRAW_PROFILE AND NOT CUBE_OUTPUT_SUFFIX MATCHES "^Research[A-Za-z0-9_]*DrawProfile$")
    message(FATAL_ERROR "Draw diagnostics require a separate Research...DrawProfile output suffix.")
endif()
if((BD_CUBE_PROFILE OR BD_CUBE_RESEARCH_SHAPE_MATRIX OR NOT BD_CUBE_VARIANT STREQUAL "Current")
    AND NOT CUBE_OUTPUT_SUFFIX MATCHES "^Research[A-Za-z0-9_]*$")
    message(FATAL_ERROR "Diagnostic and experimental builds require a Research output suffix.")
endif()
set(CUBE_NODE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../plugins/bdUtilNodes")
set(CUBE_KINDS Geometry SubScene)
if(BD_CUBE_VARIANT STREQUAL "Draw")
    list(APPEND CUBE_KINDS Draw)
endif()
foreach(kind IN LISTS CUBE_KINDS)
    add_library(bdShpCube${kind} SHARED CubePlugin.cpp "${CUBE_NODE_ROOT}/src/nodes/BdShpLCubeNode.cpp")
    target_include_directories(bdShpCube${kind} PRIVATE
        "${MAYA_LOCATION}/include" "${CUBE_NODE_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(bdShpCube${kind} PRIVATE cxx_std_17)
    target_compile_definitions(bdShpCube${kind} PRIVATE NT_PLUGIN REQUIRE_IOSTREAM)
    if(BD_CUBE_RESEARCH_SHAPE_MATRIX)
        target_compile_definitions(bdShpCube${kind} PRIVATE BD_CUBE_RESEARCH_SHAPE_MATRIX)
    endif()
    if(BD_CUBE_PROFILE)
        target_compile_definitions(bdShpCube${kind} PRIVATE BD_CUBE_PROFILE)
    endif()
    target_compile_options(bdShpCube${kind} PRIVATE /EHsc /utf-8 /permissive- /W4)
    target_link_libraries(bdShpCube${kind} PRIVATE
        "${MAYA_LOCATION}/lib/Foundation.lib"
        "${MAYA_LOCATION}/lib/OpenMaya.lib"
        "${MAYA_LOCATION}/lib/OpenMayaUI.lib"
        "${MAYA_LOCATION}/lib/OpenMayaRender.lib")
    set_target_properties(bdShpCube${kind} PROPERTIES PREFIX "" SUFFIX ".mll" CXX_EXTENSIONS OFF)
    set_target_properties(bdShpCube${kind} PROPERTIES OUTPUT_NAME "bdShpCube${kind}${CUBE_OUTPUT_SUFFIX}")
    if(NOT BD_CUBE_ARTIFACT_DIRECTORY STREQUAL "")
        set_target_properties(bdShpCube${kind} PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY_RELEASE "${BD_CUBE_ARTIFACT_DIRECTORY}")
    endif()
endforeach()
target_sources(bdShpCubeGeometry PRIVATE "${CUBE_NODE_ROOT}/src/nodes/BdShpLCubeGeometryOverride.cpp")
target_sources(bdShpCubeSubScene PRIVATE CubeSubSceneOverride.cpp)
target_compile_definitions(bdShpCubeSubScene PRIVATE BD_CUBE_SUBSCENE)
if(BD_CUBE_VARIANT STREQUAL "Draw")
    target_sources(bdShpCubeDraw PRIVATE CubeDrawOverride.cpp)
    target_compile_definitions(bdShpCubeDraw PRIVATE BD_CUBE_DRAW_OVERRIDE)
endif()
if(BD_CUBE_VARIANT STREQUAL "NodeParallel")
    target_compile_definitions(bdShpCubeGeometry PRIVATE BD_CUBE_NODE_PARALLEL)
elseif(BD_CUBE_VARIANT STREQUAL "SubSceneMatrix")
    target_compile_definitions(bdShpCubeSubScene PRIVATE BD_CUBE_SUBSCENE_MATRIX)
endif()
if(BD_CUBE_DRAW_PROFILE)
    target_compile_definitions(bdShpCubeGeometry PRIVATE BD_CUBE_DRAW_PROFILE)
endif()

if(BD_CUBE_RESEARCH_FIN_STUDY OR BD_CUBE_RESEARCH_DUMMY_BOOL_STUDY OR BD_CUBE_RESEARCH_STATIC_ATTRIBUTES_STUDY
    OR BD_CUBE_RESEARCH_DISPLAY_STATE_STUDY)
    set(nodeSources
        "${CUBE_NODE_ROOT}/src/nodes/BdShpLCubeNode.cpp"
        "${CUBE_NODE_ROOT}/src/nodes/BdShpLDeformableCubeNode.cpp")
    set(overrideSources
        "${CUBE_NODE_ROOT}/src/nodes/BdShpLCubeGeometryOverride.cpp"
        "${CUBE_NODE_ROOT}/src/nodes/BdShpLDeformableCubeSubSceneOverride.cpp")
    function(configure_shape_research target)
        target_include_directories(${target} PRIVATE
            "${MAYA_LOCATION}/include" "${CUBE_NODE_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_features(${target} PRIVATE cxx_std_17)
        target_compile_definitions(${target} PRIVATE NT_PLUGIN REQUIRE_IOSTREAM)
        target_compile_options(${target} PRIVATE /EHsc /utf-8 /permissive- /W4 /Zc:__cplusplus)
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    endfunction()

    function(configure_shape_research_plugin target)
        configure_shape_research(${target})
        target_link_libraries(${target} PRIVATE
            "${MAYA_LOCATION}/lib/Foundation.lib"
            "${MAYA_LOCATION}/lib/OpenMaya.lib"
            "${MAYA_LOCATION}/lib/OpenMayaAnim.lib"
            "${MAYA_LOCATION}/lib/OpenMayaUI.lib"
            "${MAYA_LOCATION}/lib/OpenMayaRender.lib")
        set_target_properties(${target} PROPERTIES PREFIX "" SUFFIX ".mll")
        if(NOT BD_CUBE_ARTIFACT_DIRECTORY STREQUAL "")
            set_target_properties(${target} PROPERTIES
                RUNTIME_OUTPUT_DIRECTORY_RELEASE "${BD_CUBE_ARTIFACT_DIRECTORY}")
        endif()
    endfunction()

    set(finVariants "")
    if(BD_CUBE_RESEARCH_FIN_STUDY)
        list(APPEND finVariants Fixed Attributes)
    endif()
    if(BD_CUBE_RESEARCH_DUMMY_BOOL_STUDY)
        # 属性の登録数だけを変え、ノードと描画のオブジェクトファイルを共有する。
        add_library(bdShpCubeWithFinBoolObjects OBJECT ${nodeSources} ${overrideSources})
        configure_shape_research(bdShpCubeWithFinBoolObjects)
        target_compile_definitions(bdShpCubeWithFinBoolObjects PRIVATE BD_CUBE_RESEARCH_FIN)
        list(APPEND finVariants Bool0 Bool100)
    endif()

    foreach(variant IN LISTS finVariants)
        set(target bdShpCubeWithFinResearch${variant})
        if(variant MATCHES "^Bool(0|100)$")
            add_library(${target} SHARED CubeFinPlugin.cpp $<TARGET_OBJECTS:bdShpCubeWithFinBoolObjects>)
            target_compile_definitions(${target} PRIVATE BD_CUBE_RESEARCH_DUMMY_BOOL_COUNT=${CMAKE_MATCH_1})
        else()
            add_library(${target} SHARED CubeFinPlugin.cpp ${nodeSources} ${overrideSources})
            if(variant STREQUAL "Attributes")
                target_compile_definitions(${target} PRIVATE BD_CUBE_RESEARCH_FIN_ATTRIBUTES)
            endif()
        endif()
        configure_shape_research_plugin(${target})
        target_compile_definitions(${target} PRIVATE BD_CUBE_RESEARCH_FIN)
    endforeach()

    if(BD_CUBE_RESEARCH_STATIC_ATTRIBUTES_STUDY)
        foreach(family Base Fin)
            foreach(flavor Profile Release)
                set(definitions "")
                if(family STREQUAL "Fin")
                    list(APPEND definitions BD_CUBE_RESEARCH_FIN)
                endif()
                set(suffix "")
                if(flavor STREQUAL "Profile")
                    list(APPEND definitions BD_CUBE_PROFILE)
                    set(suffix Profile)
                endif()

                # 同形状の2版は、描画とプラグイン登録のオブジェクトを共有する。
                set(common bdShpCubeStaticStudy${family}${flavor}Objects)
                add_library(${common} OBJECT CubeFinPlugin.cpp ${overrideSources})
                configure_shape_research(${common})
                target_compile_definitions(${common} PRIVATE ${definitions})
                foreach(settings Fixed Attributes)
                    set(target bdShpCubeStaticStudy${family}${settings}${suffix})
                    add_library(${target} SHARED ${nodeSources} $<TARGET_OBJECTS:${common}>)
                    configure_shape_research_plugin(${target})
                    target_compile_definitions(${target} PRIVATE ${definitions})
                    if(family STREQUAL "Base" AND settings STREQUAL "Fixed")
                        target_compile_definitions(${target} PRIVATE BD_CUBE_RESEARCH_FIXED_BASE)
                    elseif(family STREQUAL "Fin" AND settings STREQUAL "Attributes")
                        target_compile_definitions(${target} PRIVATE BD_CUBE_RESEARCH_FIN_ATTRIBUTES)
                    endif()
                endforeach()
            endforeach()
        endforeach()
    endif()

    if(BD_CUBE_RESEARCH_DISPLAY_STATE_STUDY)
        foreach(flavor Release Profile)
            set(suffix "")
            if(flavor STREQUAL "Profile")
                set(suffix Profile)
            endif()

            set(common bdShpCubeDisplayState${flavor}Objects)
            add_library(${common} OBJECT ${nodeSources}
                "${CUBE_NODE_ROOT}/src/nodes/BdShpLCubeGeometryOverride.cpp")
            configure_shape_research(${common})
            if(suffix)
                target_compile_definitions(${common} PRIVATE BD_CUBE_PROFILE)
            endif()

            foreach(variant Baseline Cache)
                set(target bdShpCubeDisplayState${variant}${suffix})
                add_library(${target} SHARED CubeFinPlugin.cpp $<TARGET_OBJECTS:${common}>
                    "${CUBE_NODE_ROOT}/src/nodes/BdShpLDeformableCubeSubSceneOverride.cpp")
                configure_shape_research_plugin(${target})
                if(suffix)
                    target_compile_definitions(${target} PRIVATE BD_CUBE_PROFILE)
                endif()
                if(variant STREQUAL "Cache")
                    target_compile_definitions(${target} PRIVATE BD_CUBE_CACHE_DISPLAY_STATE)
                    target_link_libraries(${target} PRIVATE "${MAYA_LOCATION}/lib/OpenMayaAnim.lib")
                endif()
            endforeach()
        endforeach()
    endif()
endif()
